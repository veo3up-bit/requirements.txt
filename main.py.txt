import flet as ft
import os
from google.oauth2.credentials import Credentials
from googleapiclient.discovery import build
from googleapiclient.http import MediaFileUpload

SCOPES = ['https://www.googleapis.com/auth/youtube.force-ssl']

def main(page: ft.Page):
    page.title = "YouTube Channel Bot"
    page.theme_mode = ft.ThemeMode.DARK
    page.scroll = ft.ScrollMode.AUTO
    page.padding = 20

    selected_file_path = ft.Text(value="No video selected", color=ft.colors.GREY_400)
    status_log = ft.Text(value="Status: Ready", color=ft.colors.GREEN_400)

    # Input fields for Authentication
    access_token_input = ft.TextField(
        label="OAuth Access Token",
        password=True,
        can_reveal_password=True,
        hint_text="Paste your YouTube API Access Token here"
    )

    # Input fields for Video Upload
    title_input = ft.TextField(label="Video Title")
    desc_input = ft.TextField(label="Video Description", multiline=True, min_lines=3)
    privacy_dropdown = ft.Dropdown(
        label="Privacy Status",
        value="private",
        options=[
            ft.dropdown.Option("public"),
            ft.dropdown.Option("unlisted"),
            ft.dropdown.Option("private"),
        ]
    )

    # Input fields for Auto-Reply
    video_id_input = ft.TextField(label="Target Video ID (for comments)")
    reply_text_input = ft.TextField(label="Auto-Reply Message")

    # File Picker handler
    def on_file_picked(e: ft.FilePickerResultEvent):
        if e.files:
            selected_file_path.value = e.files[0].path
            page.update()

    file_picker = ft.FilePicker(on_result=on_file_picked)
    page.overlay.append(file_picker)

    # Helper function to initialize YouTube API client
    def get_youtube_service():
        if not access_token_input.value:
            status_log.value = "Error: Please provide an Access Token first."
            status_log.color = ft.colors.RED_400
            page.update()
            return None
        creds = Credentials(token=access_token_input.value)
        return build('youtube', 'v3', credentials=creds)

    # Function to upload video
    def upload_video_action(e):
        youtube = get_youtube_service()
        if not youtube:
            return
        if selected_file_path.value == "No video selected":
            status_log.value = "Error: Please select a video file first."
            status_log.color = ft.colors.RED_400
            page.update()
            return

        try:
            status_log.value = "Uploading video... Please wait."
            status_log.color = ft.colors.YELLOW_400
            page.update()

            body = {
                'snippet': {
                    'title': title_input.value or "Uploaded via Bot",
                    'description': desc_input.value or "",
                    'categoryId': "20"
                },
                'status': {
                    'privacyStatus': privacy_dropdown.value,
                    'selfDeclaredMadeForKids': False
                }
            }

            media = MediaFileUpload(selected_file_path.value, chunksize=-1, resumable=True)
            request = youtube.videos().insert(
                part=','.join(body.keys()),
                body=body,
                media_body=media
            )
            response = request.execute()
            
            status_log.value = f"Success! Uploaded Video ID: {response['id']}"
            status_log.color = ft.colors.GREEN_400
        except Exception as ex:
            status_log.value = f"Upload Error: {str(ex)}"
            status_log.color = ft.colors.RED_400
        page.update()

    # Function to auto-reply to comments
    def reply_comments_action(e):
        youtube = get_youtube_service()
        if not youtube:
            return
        if not video_id_input.value or not reply_text_input.value:
            status_log.value = "Error: Enter Video ID and Reply Message."
            status_log.color = ft.colors.RED_400
            page.update()
            return

        try:
            status_log.value = "Checking comments..."
            page.update()

            request = youtube.commentThreads().list(
                part="snippet",
                videoId=video_id_input.value,
                maxResults=10
            )
            response = request.execute()
            replied_count = 0

            for item in response.get('items', []):
                if item['snippet']['totalReplyCount'] == 0:
                    comment_id = item['id']
                    reply_body = {
                        'snippet': {
                            'parentId': comment_id,
                            'textOriginal': reply_text_input.value
                        }
                    }
                    youtube.comments().insert(part="snippet", body=reply_body).execute()
                    replied_count += 1

            status_log.value = f"Success! Replied to {replied_count} new comments."
            status_log.color = ft.colors.GREEN_400
        except Exception as ex:
            status_log.value = f"Reply Error: {str(ex)}"
            status_log.color = ft.colors.RED_400
        page.update()

    # Build UI Layout
    page.add(
        ft.Text("YouTube Bot Manager", size=24, weight=ft.FontWeight.BOLD),
        status_log,
        ft.Divider(),
        
        ft.Text("1. Authentication", size=18, weight=ft.FontWeight.BOLD),
        access_token_input,
        ft.Divider(),

        ft.Text("2. Auto-Upload Video", size=18, weight=ft.FontWeight.BOLD),
        ft.ElevatedButton(
            "Select Video File",
            icon=ft.icons.UPLOAD_FILE,
            on_click=lambda _: file_picker.pick_files(allow_multiple=False, file_type=ft.FilePickerFileType.VIDEO)
        ),
        selected_file_path,
        title_input,
        desc_input,
        privacy_dropdown,
        ft.ElevatedButton("Start Upload", icon=ft.icons.CLOUD_UPLOAD, on_click=upload_video_action),
        ft.Divider(),

        ft.Text("3. Auto-Reply to Comments", size=18, weight=ft.FontWeight.BOLD),
        video_id_input,
        reply_text_input,
        ft.ElevatedButton("Run Auto-Reply", icon=ft.icons. SEND, on_click=reply_comments_action)
    )

ft.app(target=main)
